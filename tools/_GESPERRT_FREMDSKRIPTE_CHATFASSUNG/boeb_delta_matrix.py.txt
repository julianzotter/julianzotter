TOLERANZ_KNOTEN_MM = 0.1       # P0: Δ ≤ 0.1 mm für 7 geänderte
TOLERANZ_KRAFT_PROZENT = 1.0    # P1: Δ ≤ 1% für Kräfte
TOLERANZ_VERF_PROZENT = 5.0     # P1: Δ ≤ 5% für Verformungen

# 7 geänderte Knoten (aus transformed_nodes)

# ============================================================
# HELFER
# ============================================================
def read_csv(path):
    if not path.exists():
        log.warning(f"Fehlt: {path}")
        return []
    with open(path, encoding="utf-8") as f:
        return list(csv.DictReader(f, delimiter=";"))


def safe_float(v, default=0.0):
    try:
        return float(v) if v not in (None, "", "None") else default
    except (ValueError, TypeError):
        return default


# ============================================================
# KNOTEN-VERGLEICH
# ============================================================
def vergleiche_knoten():
    """Vergleicht RF6-Knoten mit transformed_nodes.csv."""
    log.info("Vergleiche Knotenkoordinaten...")
    transformed = {}
    for r in read_csv(INPUT / "transformed_nodes.csv"):
        try:
            transformed[int(r["RFEM"])] = (
                safe_float(r["X_m"]), safe_float(r["Y_m"]), safe_float(r["Z_m"]))
        except Exception:
            pass

    rows = []
    max_delta = 0.0
    for r in read_csv(RESULTS / "03_Knotenverformungen.csv"):
        try:
            no = int(r.get("node_no", r.get("No", 0)))
            if no in transformed:
                tx, ty, tz = transformed[no]
                dx = abs(safe_float(r.get("x")) - tx) * 1000  # m → mm
                dy = abs(safe_float(r.get("y")) - ty) * 1000
                dz = abs(safe_float(r.get("z")) - tz) * 1000
                dmax = max(dx, dy, dz)
                max_delta = max(max_delta, dmax)
                rows.append({
                    "no": no, "delta_x_mm": round(dx, 3),
                    "delta_y_mm": round(dy, 3), "delta_z_mm": round(dz, 3),
                    "delta_max_mm": round(dmax, 3),
                    "status": "PASS" if dmax <= TOLERANZ_KNOTEN_MM else "FAIL",
                })
        except Exception as e:
            log.warning(f"  Knoten: {e}")

    log.info(f"  Max Δ: {max_delta:.3f} mm "
             f"({'PASS' if max_delta <= TOLERANZ_KNOTEN_MM else 'FAIL'})")
    return rows, max_delta


# ============================================================
# KRAFT-VERGLEICH
# ============================================================
def vergleiche_kraefte():
    """Vergleicht Seilkräfte RF6 vs. RF5."""
    log.info("Vergleiche Seilkräfte...")
    rf6 = read_csv(RESULTS / "01_Seilkraefte_N.csv")
    rows = []

    for r in rf6:
        try:
            stab = int(r.get("member_no", r.get("No", 0)))
            n = safe_float(r.get("n", r.get("N", 0))) / 1000  # N → kN
            n_max = abs(n)

            # Ausnutzung gegen ZBk = 47 kN
            eta = n_max / 47.0

            rows.append({
                "stab": stab, "N_kN": round(n, 3),
                "eta_ZBk": round(eta, 3),
                "status": "PASS" if eta <= 1.0 else "BLOCK",
            })
        except Exception as e:
            log.warning(f"  Kraft: {e}")

    # Sortiere nach Ausnutzung
    rows.sort(key=lambda x: x["eta_ZBk"], reverse=True)
    log.info(f"  Max η: {rows[0]['eta_ZBk'] if rows else 0}")
    return rows


# ============================================================
# LAGER-VERGLEICH
# ============================================================
def vergleiche_lager():
    """Vergleicht Lagerreaktionen RF6 vs. RF5."""
    log.info("Vergleiche Lagerreaktionen...")
    rf6 = read_csv(RESULTS / "02_Lagerkraefte_global.csv")
    rows = []

    for r in rf6:
        try:
            no = int(r.get("node_no", r.get("No", 0)))
            px = safe_float(r.get("p_x", r.get("PX", 0))) / 1000  # N → kN
            py = safe_float(r.get("p_y", r.get("PY", 0))) / 1000
            pz = safe_float(r.get("p_z", r.get("PZ", 0))) / 1000
            pres = (px**2 + py**2 + pz**2) ** 0.5

            rows.append({
                "node": no, "P_X_kN": round(px, 3),
                "P_Y_kN": round(py, 3), "P_Z_kN": round(pz, 3),
                "P_res_kN": round(pres, 3),
            })
        except Exception as e:
            log.warning(f"  Lager: {e}")

    log.info(f"  {len(rows)} Lagerreaktionen")
    return rows


# ============================================================
# VERFORMUNGS-VERGLEICH
# ============================================================
def vergleiche_verformungen():
    """Extrahiert Verformungen an Kernknoten."""
    log.info("Vergleiche Verformungen...")
    rf6 = read_csv(RESULTS / "03_Knotenverformungen.csv")
    rows = []

    for r in rf6:
        try:
            no = int(r.get("node_no", r.get("No", 0)))
            if no not in KERNKNOTEN:
                continue
            ux = safe_float(r.get("u_x", r.get("UX", 0))) * 1000  # m → mm
            uy = safe_float(r.get("u_y", r.get("UY", 0))) * 1000
            uz = safe_float(r.get("u_z", r.get("UZ", 0))) * 1000
            ures = (ux**2 + uy**2 + uz**2) ** 0.5

            rows.append({
                "node": no, "u_X_mm": round(ux, 2),
                "u_Y_mm": round(uy, 2), "u_Z_mm": round(uz, 2),
                "u_res_mm": round(ures, 2),
                "status": "⚠ P0" if ures > 50 else "OK",  # 50mm Schwellwert
            })
        except Exception as e:
            log.warning(f"  Verformung: {e}")

    log.info(f"  {len(rows)} Kernknoten")
    return rows

# XLSX-EXPORT
# ============================================================
def export_xlsx(delta_knoten, kraefte, lager, verformungen):
    """Erstellt XLSX mit mehreren Sheets."""
    try:
        import pandas as pd
    except ImportError:
        log.warning("pandas fehlt – XLSX-Export übersprungen")
        return None

    path = RESULTS / "DELTA_MATRIX.xlsx"
    with pd.ExcelWriter(path, engine="openpyxl") as writer:
        pd.DataFrame(delta_knoten).to_excel(
            writer, sheet_name="Delta_Knoten", index=False)
        pd.DataFrame(kraefte).to_excel(
            writer, sheet_name="Seilkraefte", index=False)
        pd.DataFrame(lager).to_excel(
            writer, sheet_name="Lagerkraefte", index=False)
        pd.DataFrame(verformungen).to_excel(
            writer, sheet_name="Kernknoten_Verformungen", index=False)

    log.info(f"  XLSX: {path}")
    return path
